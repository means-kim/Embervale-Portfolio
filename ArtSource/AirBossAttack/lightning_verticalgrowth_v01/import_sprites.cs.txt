string path = "Assets/04.Sprites/AirBossAttack/Attack_Lightning_VerticalGrowth_AirBoss_32F_v01.png";
string source = "ArtSource/AirBossAttack/lightning_verticalgrowth_v01/Attack_Lightning_VerticalGrowth_AirBoss_32F_v01_Final.png";
if (System.IO.File.Exists(path)) throw new System.Exception("Existing asset protected: " + path);
System.IO.File.Copy(source, path, false);
AssetDatabase.ImportAsset(path, ImportAssetOptions.ForceSynchronousImport);
var importer = (TextureImporter)AssetImporter.GetAtPath(path);
importer.textureType = TextureImporterType.Sprite;
importer.spriteImportMode = SpriteImportMode.Multiple;
importer.spritePixelsPerUnit = 256;
importer.alphaSource = TextureImporterAlphaSource.FromInput;
importer.alphaIsTransparency = true;
importer.mipmapEnabled = false;
importer.npotScale = TextureImporterNPOTScale.None;
importer.maxTextureSize = 4096;
importer.textureCompression = TextureImporterCompression.Uncompressed;
importer.filterMode = FilterMode.Bilinear;
importer.wrapMode = TextureWrapMode.Clamp;
var settings = new TextureImporterSettings();
importer.ReadTextureSettings(settings);
settings.spriteMeshType = SpriteMeshType.FullRect;
settings.spriteGenerateFallbackPhysicsShape = false;
importer.SetTextureSettings(settings);
importer.SaveAndReimport();
var factory = new UnityEditor.U2D.Sprites.SpriteDataProviderFactories();
factory.Init();
var provider = factory.GetSpriteEditorDataProviderFromObject(importer);
provider.InitSpriteEditorDataProvider();
var rects = new UnityEditor.SpriteRect[32];
var pairs = new System.Collections.Generic.List<UnityEditor.SpriteNameFileIdPair>();
for (int f = 0; f < 32; f++) {
  var id = GUID.Generate();
  rects[f] = new UnityEditor.SpriteRect {
    name = "Attack_Lightning_VerticalGrowth_AirBoss_v01_" + f.ToString("D2"),
    rect = new Rect((f % 8) * 384, (3 - f / 8) * 768, 384, 768),
    alignment = SpriteAlignment.Center, pivot = new Vector2(0.5f, 0.5f), spriteID = id
  };
  pairs.Add(new UnityEditor.SpriteNameFileIdPair(rects[f].name, id));
}
provider.SetSpriteRects(rects);
provider.GetDataProvider<UnityEditor.U2D.Sprites.ISpriteNameFileIdDataProvider>().SetNameFileIdPairs(pairs);
provider.Apply();
importer.SaveAndReimport();
var texture = AssetDatabase.LoadAssetAtPath<Texture2D>(path);
var sprites = AssetDatabase.LoadAllAssetsAtPath(path).OfType<Sprite>().OrderBy(s=>s.name).ToArray();
if (sprites.Length != 32 || texture.width != 3072 || texture.height != 3072) throw new System.Exception("Wrong sprite count or imported texture size.");
for (int f=0; f<32; f++) {
  var sprite=sprites[f];
  if(sprite.rect != new Rect((f%8)*384,(3-f/8)*768,384,768) || sprite.pivot != new Vector2(192,384) || sprite.pixelsPerUnit != 256) throw new System.Exception("Sprite rect/pivot/PPU mismatch: " + sprite.name);
}
var report = new {
  path, guid=AssetDatabase.AssetPathToGUID(path),
  textureWidth=texture.width,textureHeight=texture.height,
  count=sprites.Length, cellWidth=384,cellHeight=768,
  pivot="0.5,0.5",pivotPixels="192,384",ppu=256,
  filter=importer.filterMode.ToString(),compression=importer.textureCompression.ToString(),
  generatePhysicsShape=settings.spriteGenerateFallbackPhysicsShape,
  scene=UnityEngine.SceneManagement.SceneManager.GetActiveScene().path,
  sceneDirty=UnityEngine.SceneManagement.SceneManager.GetActiveScene().isDirty,
  sprites=sprites.Select(s=>new{name=s.name,rect=s.rect.ToString(),pivot=s.pivot.ToString()}).ToArray()
};
System.IO.File.WriteAllText("ArtSource/AirBossAttack/lightning_verticalgrowth_v01/unity_validation.json", Newtonsoft.Json.JsonConvert.SerializeObject(report,Newtonsoft.Json.Formatting.Indented));
return report;
