using UnityEngine;

public class BossAttackController : MonoBehaviour
{
    [SerializeField] private Transform _target;
    [SerializeField] private BossSlamAttack _slamAttack;
    [SerializeField] private BossSwingAttack _swingAttack;
    [SerializeField] private BossChargeAttack _chargeAttack;
    [SerializeField] private BossIceThornAttack _iceThornAttack;
    [SerializeField] private BossAttackDataSO _attackDataSo;
    [SerializeField] private BossAttackSellector _bossAttackSellector;
    [SerializeField] private BossHit _bossHit;

    [SerializeField] private BossMovement _movement;
    [SerializeField] private BossFacing _facing;
    [SerializeField] private Health _health;
    [SerializeField] private Animator _animator;

    private BossAttack _currentAttack;
    public bool IsAttacking => _currentAttack != null;

    private void Start()
    {
        AttackDataSettings();
    }

    private void Update()
    {
        if (_health.IsDead) return;
        if (_currentAttack != null) return;
        if (_bossHit.IsHit) return;
        if (_target == null) return;

        float distance = Mathf.Abs(_target.position.x - transform.position.x);
        BossAttack sellectAttack = _bossAttackSellector.SellectAttack(distance);

        if (sellectAttack == null) return;

        BeginAttack(sellectAttack);
    }

    // Explicit entry point until the Phase 2 selector is implemented.
    public bool TryStartIceThornAttack()
    {
        if (!isActiveAndEnabled || _health == null || _health.IsDead || _currentAttack != null) return false;
        if (_bossHit != null && _bossHit.IsHit) return false;
        if (_target == null || _iceThornAttack == null) return false;
        var state = _animator.GetCurrentAnimatorStateInfo(0);
        if (_animator.IsInTransition(0) || (!state.IsName("Boss_Idle") && !state.IsName("Boss_Walk"))) return false;
        AttackDataSettings();
        if (!_iceThornAttack.CanAttack(Mathf.Abs(_target.position.x - transform.position.x))) return false;
        BeginAttack(_iceThornAttack);
        return true;
    }

    private void BeginAttack(BossAttack attack)
    {
        _currentAttack = attack;
        _currentAttack.OnAttackFinished += HandleAttackFinsed;
        _currentAttack.StartAttack();

        _movement.enabled = false;
        _facing.enabled = false;
        _animator.SetBool("IsWalk", false);
    }

    private void OnDisable()
    {
        CancelCurrentAttack();

        if (_health != null && !_health.IsDead)
        {
            _movement.enabled = true;
            _facing.enabled = true;
        }
    }

    private void HandleAttackFinsed()
    {
        CancelCurrentAttack();


        if (_health.IsDead) return;

        _movement.enabled = true;
        _facing.enabled = true;
    }

    public void CancelCurrentAttack()
    {
        if (_currentAttack == null) return;

        _currentAttack.OnAttackFinished -= HandleAttackFinsed;
        _currentAttack.CancelAttack();
        _currentAttack = null;
    }

    private void AttackDataSettings()
    {
        foreach (BossAttackData data in _attackDataSo.Attacks)
        {
            if (data == null) continue;

            switch (data.Type)
            {
                case BossAttackType.Slam:
                    _slamAttack.SetData(data);
                    break;

                case BossAttackType.Swing:
                    _swingAttack.SetData(data);
                    break;

                case BossAttackType.Charge:
                    _chargeAttack.SetData(data);
                    break;

                case BossAttackType.IceThorn:
                    if (_iceThornAttack != null) _iceThornAttack.SetData(data);
                    break;
            }
        }
    }
}
