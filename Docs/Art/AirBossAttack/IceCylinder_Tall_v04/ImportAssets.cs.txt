
string texturePath="Assets/04.Sprites/AirBossAttack/Attack_IceCylinder_Tall_AirBoss_24F_v04.png";
string clipPath="Assets/05.Animation/AirBossAttack/Attack_LargeIcePillar_White_Tall_v04.anim";
string sourcePath="Assets/05.Animation/AirBossAttack/Attack_LargeIceIcePillar_White.anim";
if(UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.AnimationClip>(clipPath)!=null) throw new System.Exception("Destination clip exists.");
UnityEditor.AssetDatabase.ImportAsset(texturePath,UnityEditor.ImportAssetOptions.ForceSynchronousImport);
var importer=(UnityEditor.TextureImporter)UnityEditor.AssetImporter.GetAtPath(texturePath);
importer.textureType=UnityEditor.TextureImporterType.Sprite;
importer.spriteImportMode=UnityEditor.SpriteImportMode.Multiple;
importer.spritePixelsPerUnit=512;
importer.maxTextureSize=4096;
importer.npotScale=UnityEditor.TextureImporterNPOTScale.None;
importer.alphaIsTransparency=true;
importer.mipmapEnabled=false;
importer.isReadable=false;
importer.filterMode=UnityEngine.FilterMode.Bilinear;
importer.wrapMode=UnityEngine.TextureWrapMode.Clamp;
importer.textureCompression=UnityEditor.TextureImporterCompression.Uncompressed;
var settings=new UnityEditor.TextureImporterSettings();
importer.ReadTextureSettings(settings);
settings.spriteMeshType=UnityEngine.SpriteMeshType.FullRect;
settings.spriteAlignment=(int)UnityEngine.SpriteAlignment.Center;
settings.spritePivot=new UnityEngine.Vector2(0.5f,0.5f);
settings.spriteGenerateFallbackPhysicsShape=false;
importer.SetTextureSettings(settings);
var platform=importer.GetDefaultPlatformTextureSettings();
platform.maxTextureSize=4096;platform.textureCompression=UnityEditor.TextureImporterCompression.Uncompressed;
importer.SetPlatformTextureSettings(platform);
var standalone=importer.GetPlatformTextureSettings("Standalone");
standalone.overridden=false;standalone.maxTextureSize=4096;
importer.SetPlatformTextureSettings(standalone);
var metas=new UnityEditor.SpriteMetaData[24];
for(int i=0;i<24;i++) metas[i]=new UnityEditor.SpriteMetaData {name="Attack_IceCylinder_Tall_v04_"+(i+1).ToString("D2"),rect=new UnityEngine.Rect((i%6)*512,4096-((i/6)+1)*1024,512,1024),alignment=(int)UnityEngine.SpriteAlignment.Center,pivot=new UnityEngine.Vector2(0.5f,0.5f)};
importer.spritesheet=metas;
importer.SaveAndReimport();
var sprites=UnityEditor.AssetDatabase.LoadAllAssetsAtPath(texturePath).OfType<UnityEngine.Sprite>().OrderBy(s=>s.name).ToArray();
if(sprites.Length!=24)throw new System.Exception("Expected 24 sprites, got "+sprites.Length);
var source=UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.AnimationClip>(sourcePath);
var clip=UnityEngine.Object.Instantiate(source);
clip.name="Attack_LargeIcePillar_White_Tall_v04";
var bindings=UnityEditor.AnimationUtility.GetObjectReferenceCurveBindings(source);
var binding=bindings.Single(b=>b.type==typeof(UnityEngine.SpriteRenderer)&&b.propertyName=="m_Sprite");
var keys=UnityEditor.AnimationUtility.GetObjectReferenceCurve(source,binding);
if(keys.Length!=25)throw new System.Exception("Unexpected source keys");
for(int i=0;i<24;i++)keys[i].value=sprites[i];
keys[24].value=null;
UnityEditor.AnimationUtility.SetObjectReferenceCurve(clip,binding,keys);
UnityEditor.AssetDatabase.CreateAsset(clip,clipPath);
UnityEditor.AssetDatabase.SaveAssetIfDirty(clip);
var tex=UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.Texture2D>(texturePath);
return new {texture=texturePath,dimensions=new[]{tex.width,tex.height},count=sprites.Length,ppu=sprites[0].pixelsPerUnit,pivot=sprites[0].pivot,rect=sprites[0].rect,clip=clipPath,length=clip.length,frameRate=clip.frameRate,events=UnityEditor.AnimationUtility.GetAnimationEvents(clip).Select(e=>new {e.time,e.functionName}).ToArray(),loop=UnityEditor.AnimationUtility.GetAnimationClipSettings(clip).loopTime};
